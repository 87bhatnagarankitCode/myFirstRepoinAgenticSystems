-- all 10 select statements:
SELECT U.NAME,O.AMOUNT FROM USERS U INNER JOIN ORDERS O ON O.USER_ID = U.USER_ID ;

SELECT U.NAME,COALESCE(SUM(O.AMOUNT),0) AS TOTAL_ORDER FROM USERS U INNER JOIN ORDERS O ON O.USER_ID = U.USER_ID GROUP BY U.NAME  ;

SELECT U.NAME, A.LOGIN_COUNTS FROM USERS U LEFT JOIN ACTIVITY A ON U.USER_ID = A.USER_ID ;

SELECT U.NAME, A.LOGIN_COUNTS FROM USERS U LEFT JOIN ACTIVITY A ON U.USER_ID = A.USER_ID WHERE A.LOGIN_COUNTS IS NULL  ;

SELECT U.USER_ID,U.NAME,COALESCE(SUM(O.AMOUNT),0) AS TOTAL_ORDER_AMOUNT,COALESCE(SUM(A.LOGIN_COUNTS),0) AS LOGIN_COUNT FROM USERS U LEFT JOIN ORDERS O ON 
U.USER_ID = O.USER_ID LEFT JOIN ACTIVITY A ON U.USER_ID = A.USER_ID GROUP BY U.NAME ;

SELECT U.NAME,COALESCE(SUM(O.AMOUNT),0) AS TOTAL_ORDERS FROM USERS U INNER JOIN ORDERS O ON O.USER_ID = U.USER_ID WHERE O.AMOUNT >400 GROUP BY U.NAME; 

SELECT U.NAME,COALESCE(SUM(O.AMOUNT),0) AS TOTAL_ORDERS FROM USERS U INNER JOIN ORDERS O ON O.USER_ID = U.USER_ID GROUP BY U.NAME ORDER BY O.AMOUNT  DESC LIMIT 3; 

SELECT U.USER_ID,U.NAME,COALESCE(SUM(O.AMOUNT),0) AS TOTAL_ORDER_AMOUNT,COALESCE(SUM(A.LOGIN_COUNTS),0) AS LOGIN_COUNT FROM USERS U LEFT JOIN ORDERS O ON 
U.USER_ID = O.USER_ID LEFT JOIN ACTIVITY A ON U.USER_ID = A.USER_ID GROUP BY U.NAME  ORDER BY O.AMOUNT DESC,  A.LOGIN_COUNTS DESC  ;


SELECT  U.USER_ID, U.NAME,  AVG(O.AMOUNT) AS AVG_ORDER_AMOUNT FROM USERS U LEFT JOIN ORDERS O  ON U.USER_ID = O.USER_ID GROUP BY U.USER_ID, U.NAME;

SELECT U.USER_ID,U.NAME,COALESCE(SUM(O.AMOUNT),0) AS TOTAL_ORDER_AMOUNT,COALESCE(SUM(A.LOGIN_COUNTS),0) AS LOGIN_COUNT FROM USERS U LEFT JOIN ORDERS O ON 
U.USER_ID = O.USER_ID LEFT JOIN ACTIVITY A ON U.USER_ID = A.USER_ID GROUP BY U.NAME  ORDER BY O.AMOUNT DESC LIMIT 3 ;
